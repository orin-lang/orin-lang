module std:json/number
  @Sealed
  type JsonNumber = { token: string }
    token: string
  type NumberError = { kind: NumberErrorKind; offset: Option<usize> }
    kind: NumberErrorKind
    offset: Option<usize>
  type NumberErrorKind = InvalidNumber | NumberLimit | InvalidLimits | NotIntegral | IntegerRange | NonFinite | FloatOverflow | FloatUnderflow
    InvalidNumber: NumberErrorKind
    NumberLimit: NumberErrorKind
    InvalidLimits: NumberErrorKind
    NotIntegral: NumberErrorKind
    IntegerRange: NumberErrorKind
    NonFinite: NumberErrorKind
    FloatOverflow: NumberErrorKind
    FloatUnderflow: NumberErrorKind
  function jsonNumber(text: string, maxBytes: usize): Result<JsonNumber, NumberError>
  @Ensures(result.isOk() ==> result.unwrap().token == text && text.length <= maxBytes)
  @Ensures(result.isErr()     ==> result.unwrapErr().offset.isNone() || result.unwrapErr().offset.unwrap() <= text.length)
  function numberAsInt(n: JsonNumber): Result<int, NumberError>
  @Ensures(result.isErr() ==> result.unwrapErr().offset.isNone())
  function numberErrorMessage(e: NumberError): string
  @Ensures(result.length > 0)
  function numberFromInt(n: int): JsonNumber
  @Ensures(result.token == n.toString())
