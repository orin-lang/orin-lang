    kind: FsErrorKind
    op: FsOp
    path: string
    reason: string

    NotFound: FsErrorKind
    PermissionDenied: FsErrorKind
    IsDirectory: FsErrorKind
    NotDirectory: FsErrorKind
    AlreadyExists: FsErrorKind
    Denied: FsErrorKind
    OutsideScope: FsErrorKind
    ReadOnly: FsErrorKind
    WrongMode: FsErrorKind
    TooManyFiles: FsErrorKind
    Io: FsErrorKind
  type FsOp = OpRead | OpWrite | OpOpen | OpList | OpStat | OpClose
    OpRead: FsOp
    OpWrite: FsOp
    OpOpen: FsOp
    OpList: FsOp
    OpStat: FsOp
    OpClose: FsOp
